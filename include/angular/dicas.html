BOOTSTRAP

No angular também é possivel inserir o bootstrap.

Instalando

$ npm install bootstrap@4.1.1

Carregando no APP

Esta indicação é feita em angular.json, que possui uma chave denominada build, dentro do qual está styles. Alteraremos e salvaremos seu conteúdo para incluir o caminho completo do Bootstrap na pasta "node_modules": 
"styles": [
    "src/styles.css",
    "./node_modules/bootstrap/dist/css/bootstrap.min.css"
],
"scripts": []

alterar tipo de arquivo de estilos

Alterar o padrão para novos componentes

Primeiro, informe ao Angular CLI que, a partir de agora, todo novo componente deve ser gerado com .scss. Execute este comando no terminal: 
$ ng config schematics.@schematics/angular:component.styleext scss

Atualizar o arquivo angular.json

Você precisa avisar ao Angular onde encontrar o arquivo de estilos globais (que você vai renomear em seguida). No seu angular.json, procure pelas referências a src/styles.css e mude para src/styles.scss.

Geralmente existem dois lugares: em build e em test.

Renomear os arquivos físico

Voce pode renomear todos os arquivos de uma vez (no Linux/macOS ou Git Bash), manualmente ou usar o comando abaixo: 
$ find src/ -name "*.css" -exec bash -c 'mv "$0" "${0%.css}.scss"' {} \;
Após essas mudanças, pare o servidor (ng serve) e inicie-o novamente para que o Angular reconheça as novas extensões e o pré-processador SASS.

Atualizar as referências nos Componentes

Em cada arquivo .ts dos seus componentes, você precisará atualizar o metadado styleUrls (ou styleUrl nas versões mais recentes): 
  @Component({
    selector: 'app-root',
    templateUrl: './app.component.html',
    styleUrl: './app.component.scss' // Altere de .css para .scss
  })

IDENTIFICAR INTERACAO HUMANA

import { Component, HostListener } from '@angular/core';

@Component({
  selector: 'app-human-interaction',
  template: `
    <p>Interações detectadas: {{interactionCount}}</p>
    <button [disabled]="!humanInteracted">Enviar</button>
  `
})

export class HumanInteractionComponent {
  interactionCount = 0;
  humanInteracted = false;

  private readonly INTERACTION_THRESHOLD = 5;
  
  // 🎯 Evento de movimento do mouse
  @HostListener('document:mousemove', ['$event'])
  onMouseMove(event: MouseEvent) {
    this.registerInteraction();
  }
  
  // 🎯 Evento de clique com mouse
  @HostListener('document:click', ['$event'])
  onClick(event: MouseEvent) {
    this.registerInteraction();
  }
  
  // 🎯 Eventos de toque (mobile)
  @HostListener('document:touchstart', ['$event'])
  onTouchStart(event: TouchEvent) {
    this.registerInteraction();
  }

  @HostListener('document:touchmove', ['$event'])
  onTouchMove(event: TouchEvent) {
    this.registerInteraction();
  }

  @HostListener('document:touchend', ['$event'])
  onTouchEnd(event: TouchEvent) {
    this.registerInteraction();
  }

  // 🎯 Evento de teclado
  @HostListener('document:keydown', ['$event'])
  onKeyDown(event: KeyboardEvent) {
    this.registerInteraction();
  }

  private registerInteraction() {
    if (!this.humanInteracted) {
      this.interactionCount++;
      if (this.interactionCount >= this.INTERACTION_THRESHOLD) {
        this.humanInteracted = true;
        console.log('✅ Interação humana detectada');
      }
    }
  }
}

IDENTIFICAR OUTRAS INTERACOES

import { Component, OnInit, OnDestroy } from '@angular/core';

@Component({
  selector: 'app-signin-matricula',
  templateUrl: './signin.matricula.component.html',
  styleUrls: ['./signin.matricula.component.scss']
})
export class SigninMatriculaComponent implements OnInit, OnDestroy {
  interacaoCount = 0;

  private listeners: Array<() => void> = [];

  ngOnInit() {
    // Clique em qualquer lugar
    const clickListener = () => this.incrementarInteracao();
    window.addEventListener('click', clickListener);

    // Foco em qualquer input
    const focusListener = (e: Event) => {
      if ((e.target as HTMLElement).tagName === 'INPUT') {
        this.incrementarInteracao();
      }
    };
    window.addEventListener('focusin', focusListener);

    // Arrastar com mouse
    const mouseMoveListener = () => this.incrementarInteracao();
    window.addEventListener('mousemove', mouseMoveListener);

    // Scroll (desktop e mobile)
    const scrollListener = () => this.incrementarInteracao();
    window.addEventListener('scroll', scrollListener);

    // Touch (mobile)
    const touchListener = () => this.incrementarInteracao();
    window.addEventListener('touchmove', touchListener);

    // Salva para remover depois
    this.listeners = [
      () => window.removeEventListener('click', clickListener),
      () => window.removeEventListener('focusin', focusListener),
      () => window.removeEventListener('mousemove', mouseMoveListener),
      () => window.removeEventListener('scroll', scrollListener),
      () => window.removeEventListener('touchmove', touchListener),
    ];
  }

  ngOnDestroy() {
    // Remove todos os listeners ao destruir o componente
    this.listeners.forEach(remove => remove());
  }

  incrementarInteracao() {
    this.interacaoCount++;
    // console.log('Interações:', this.interacaoCount);
  }
}
