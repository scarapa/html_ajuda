CLASSE VALIDADORA

import { Injectable } from '@angular/core';
import { FormGroup } from '@angular/forms';

@Injectable({ providedIn: 'root' })

export class GenericFormValidatorService {

  constructor(private validationMessages: ValidationMessages) { }

  processarMensagens(container: FormGroup): { [key: string]: string } {
    let messages:any = {};
    for (let controlKey in container.controls) {
      if (container.controls.hasOwnProperty(controlKey)) {
        let c = container.controls[controlKey];

        if (c instanceof FormGroup) {
          let childMessages = this.processarMensagens(c);
          Object.assign(messages, childMessages);
        } else {
          if (this.validationMessages[controlKey]) {
            messages[controlKey] = '';
            if ( (c.dirty || c.touched ) && c.errors) {
              Object.keys(c.errors).map(messageKey => {
                if (this.validationMessages[controlKey][messageKey]) {
                  messages[controlKey] += this.validationMessages[controlKey][messageKey] + '<br />';
                }
              });
            }
          }
        }
      }
    }
    return messages;
  }
}

export interface DisplayMessage {
  [key: string]: string
}
export interface ValidationMessages {
  [key: string]: { [key: string]: string } 
}

  import { AfterViewInit, Component, ElementRef, OnInit, ViewChildren } from '@angular/core';
  import { FormBuilder, FormControl, FormControlName, FormGroup, Validators } from '@angular/forms';
  import { UsuarioModel } from 'src/app/models/usuario-model';
  import { NgBrazilValidators , MASKS } from 'ng-brazil';
  import { DisplayMessage, GenericFormValidatorService, ValidationMessages } from 'src/app/services/generic-form-validator.service';
  import { CustomValidators } from 'ng2-validation';
  import { Observable,fromEvent,merge } from 'rxjs';
  
  @Component({
    selector: 'app-cadastro-form',
    templateUrl: './cadastro-form.component.html',
    styleUrls: ['./cadastro-form.component.css'],
  })
  export class CadastroFormComponent implements OnInit, AfterViewInit {
  
    cadastroForm !: FormGroup;
    usuario ? : UsuarioModel;
    formResult:string = '';
  
    validationMessages! : ValidationMessages;
    genericValidator! : GenericFormValidatorService;
    displayMessage : DisplayMessage = {};
  
    // PEGAR OS FORMS EM TELA
    @ViewChildren( FormControlName , {read:ElementRef} ) formInputElements! : ElementRef[];
  
    constructor(
      private fb: FormBuilder
    ) { 
      this.validationMessages = {
        nome : {
          required:'nome requerido',
          minlength:'minimo de 2 caracteres',
        }, 
        cpf : {
          required:'cpf requerido',
          cpf:'formato inválido',
        },
        email : {
          required:'email requerido',
          email:'formato inválido',
        },
        senha : {
          required:'senha requerida',
          rageLength:'entre 6 e 15 caracteres',
        },      
        senhaConfirmacao : {
          required:'informe senha requerido',
          rageLength:'entre 6 e 15 caracteres',
          equalTo:'senhas não conferem',
        },          
      }
  
      this.genericValidator = new GenericFormValidatorService( this.validationMessages );
    }
  
    // DEPOIS DE CONCLUIDO O CARREGAMENTO
    ngAfterViewInit() {
      let controlBlurs :Observable<any>[] ;
      controlBlurs = this.formInputElements
        .map( (formControl:ElementRef) => fromEvent( formControl.nativeElement, 'blur') );
  
        merge( ... controlBlurs).subscribe( () => {
          this.displayMessage = this.genericValidator.processarMensagens( this.cadastroForm )
        })
    }
  
    ngOnInit() {
      let senha = new FormControl('',[Validators.required, CustomValidators.rangeLength([6,15]) ]);
      let senhaConfirmacao = new FormControl('',[ Validators.required, CustomValidators.rangeLength([6,15]), CustomValidators.equalTo(senha) ]);
      this.cadastroForm = this.fb.group({});
      this.cadastroForm = this.fb.group({
        nome: ['',[Validators.required,Validators.minLength(2)]],
        cpf: ['',[<any>Validators.required,<any>NgBrazilValidators.cpf]],
        email: ['',Validators.email],
        senha: senha,
        senhaConfirmacao: senhaConfirmacao,
      });
    }
  
    adicionarUsuario(){
      // if( !this.cadastroForm.dirty && !this.cadastroForm.valid ) return ;
      let x = this.cadastroForm.value;
      // MAPEANDO DO OBJETO
      this.usuario = Object.assign({} , this.usuario , this.cadastroForm.value )
      this.formResult = JSON.stringify( this.usuario );
    }
  }
